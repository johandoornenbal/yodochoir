cmake_minimum_required(VERSION 3.22)
if(CMAKE_HOST_APPLE)
  set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Apple Silicon release architecture")
  if(NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
    message(FATAL_ERROR "Release builds support native Apple Silicon arm64")
  endif()
  set(CMAKE_OSX_DEPLOYMENT_TARGET 11.0 CACHE STRING "Minimum macOS version")
endif()
project(YodoChoir VERSION 0.6.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_library(rubberband_vendor STATIC third_party/rubberband/single/RubberBandSingle.cpp)
set_target_properties(rubberband_vendor PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(rubberband_vendor PUBLIC third_party/rubberband)
if(APPLE)
  target_link_libraries(rubberband_vendor PUBLIC "-framework Accelerate")
endif()
add_library(choir_engine Source/ChoirEngine.cpp Source/kWoodRoomEngine.cpp Source/OctaveShift.cpp)
target_include_directories(choir_engine SYSTEM PRIVATE third_party/signalsmith-stretch third_party/signalsmith-linear/include)
target_link_libraries(choir_engine PRIVATE rubberband_vendor)
set_target_properties(choir_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(choir_engine PUBLIC Source)
if(BUILD_TESTING)
  add_executable(engine_tests tests/EngineTests.cpp)
  target_link_libraries(engine_tests PRIVATE choir_engine)
  add_test(NAME ensemble_dsp COMMAND engine_tests)
  add_executable(octave_tests tests/OctaveTests.cpp)
  target_link_libraries(octave_tests PRIVATE choir_engine)
  add_test(NAME octave_pitch_engines COMMAND octave_tests)
  add_executable(woodroom_reference tests/ReferencekWoodRoom.cpp)
  target_include_directories(woodroom_reference PRIVATE tests/shim)
  target_link_libraries(woodroom_reference PRIVATE choir_engine)
  add_test(NAME unchanged_kWoodRoom COMMAND woodroom_reference)
endif()
option(YODOCHOIR_DSP_ONLY "Build portable engine and tests without JUCE" OFF)
if(YODOCHOIR_DSP_ONLY)
  return()
endif()
set(JUCE_PATH "" CACHE PATH "Optional JUCE 8.0.12 checkout")
if(JUCE_PATH)
  add_subdirectory("${JUCE_PATH}" "${CMAKE_BINARY_DIR}/JUCE")
else()
  include(FetchContent)
  FetchContent_Declare(JUCE URL https://codeload.github.com/juce-framework/JUCE/tar.gz/29396c22c93392d6738e021b83196283d6e4d850)
  FetchContent_MakeAvailable(JUCE)
endif()
juce_add_plugin(YodoChoir
  COMPANY_NAME "Yodo Audio" BUNDLE_ID "nl.doornenbal.yodochoir"
  PLUGIN_MANUFACTURER_CODE Jdau PLUGIN_CODE Ychr
  FORMATS VST3 PRODUCT_NAME "YodoChoir"
  IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE
  IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE
  COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Tools)
juce_add_binary_data(YodoChoirAssets SOURCES resources/yipc_logo.png
  NAMESPACE YodoChoirAssets HEADER_NAME YodoChoirAssets.h)
set_target_properties(YodoChoirAssets PROPERTIES POSITION_INDEPENDENT_CODE ON)
juce_generate_juce_header(YodoChoir)
target_sources(YodoChoir PRIVATE Source/Plugin.cpp Source/ReaperBridge.cpp)
target_compile_definitions(YodoChoir PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(YodoChoir PRIVATE YodoChoirAssets choir_engine juce::juce_audio_utils
  PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
if(BUILD_TESTING)
  add_executable(plugin_checks tests/PluginChecks.cpp)
  target_include_directories(plugin_checks PRIVATE "${JUCE_MODULES_DIR}")
  target_compile_definitions(plugin_checks PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
  target_link_libraries(plugin_checks PRIVATE YodoChoir)
  add_test(NAME plugin_workflow COMMAND plugin_checks)
  add_executable(recording_checks tests/RecordingChecks.cpp)
  target_include_directories(recording_checks PRIVATE "${JUCE_MODULES_DIR}")
  target_compile_definitions(recording_checks PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
  target_link_libraries(recording_checks PRIVATE YodoChoir)
  add_test(NAME track_recording COMMAND recording_checks)
  add_executable(transport_checks tests/TransportChecks.cpp)
  target_include_directories(transport_checks PRIVATE "${JUCE_MODULES_DIR}" "${JUCE_MODULES_DIR}/juce_audio_processors/format_types/VST3_SDK" "${JUCE_MODULES_DIR}/juce_audio_processors_headless/format_types/VST3_SDK")
  target_compile_definitions(transport_checks PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
  target_link_libraries(transport_checks PRIVATE YodoChoir)
  add_test(NAME transport_and_track_write COMMAND transport_checks)
  add_executable(resolution_checks tests/ResolutionChecks.cpp)
  target_include_directories(resolution_checks PRIVATE "${JUCE_MODULES_DIR}")
  target_compile_definitions(resolution_checks PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
  target_link_libraries(resolution_checks PRIVATE YodoChoir)
  add_test(NAME project_resolution_and_tweaks COMMAND resolution_checks)
  if(UNIX)
    add_executable(vst3_bus_checks tests/Vst3BusChecks.cpp)
    target_include_directories(vst3_bus_checks PRIVATE "${JUCE_MODULES_DIR}/juce_audio_processors/format_types/VST3_SDK" "${JUCE_MODULES_DIR}/juce_audio_processors_headless/format_types/VST3_SDK")
    target_link_libraries(vst3_bus_checks PRIVATE ${CMAKE_DL_LIBS})
    if(APPLE)
      target_link_libraries(vst3_bus_checks PRIVATE "-framework CoreFoundation")
    endif()
    add_dependencies(vst3_bus_checks YodoChoir_VST3)
    add_test(NAME vst3_input_pin_negotiation COMMAND vst3_bus_checks "$<TARGET_FILE:YodoChoir_VST3>")
  endif()
  add_test(NAME installer_script COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/InstallerChecks.sh")
endif()
